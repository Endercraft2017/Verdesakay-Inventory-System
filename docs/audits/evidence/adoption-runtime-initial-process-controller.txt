import concurrent.futures
import datetime
import hashlib
import json
import os
from pathlib import Path
import selectors
import signal
import sqlite3
import subprocess
import tempfile

root = Path.cwd()
out = root/'work/browser-ci'
versions = {'24.22.1': root/'work/runtime-support/node-v24.22.1-linux-x64/bin/node',
            '22.23.4': root/'work/runtime-support/node-v22.23.4-linux-x64/bin/node'}
def hashfile(p): return hashlib.sha256(p.read_bytes()).hexdigest()
def fixture():
    p = Path(tempfile.mkdtemp(prefix='adoption-process-control-'))
    with sqlite3.connect(p/'source.db') as db:
        db.executescript("CREATE TABLE users(id INTEGER PRIMARY KEY, username TEXT, password_hash TEXT); INSERT INTO users VALUES(1,'synthetic-process-owner','synthetic-original-hash'); CREATE TABLE records(id INTEGER PRIMARY KEY, data TEXT); INSERT INTO records VALUES(1,'synthetic-original-history');")
    (p/'uploads/nested').mkdir(parents=True)
    (p/'uploads/a.bin').write_bytes(bytes([0,255,16,9]));(p/'uploads/nested/b.bin').write_text('synthetic nested process upload')
    return p
def start(node, p, mode):
    env = {**os.environ, 'PATH':str(node.parent)+':'+os.environ['PATH'], 'NODE_ENV':'test',
           'DB_PATH':str(p/'source.db'), 'DATA_DIR':str(p/'tenants'), 'UPLOAD_DIR':str(p/'uploads'),
           'PLATFORM_DB_PATH':str(p/'platform.db'), 'ADOPTION_PROCESS_MODE':mode}
    return subprocess.Popen([str(node),'--import','tsx','work/browser-ci/copy-process-worker.ts'],
            cwd=root, env=env, stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
def await_barrier(child, stage):
    with selectors.DefaultSelector() as selector:
        selector.register(child.stdout, selectors.EVENT_READ)
        if not selector.select(30): raise AssertionError('Barrier timed out')
        line = child.stdout.readline()
    data = json.loads(line)
    assert data == {'event':'barrier','stage':stage}, data
    return line.decode()
def collect(child, prefix=''):
    stdout,stderr = child.communicate(timeout=30)
    return {'exit_code':child.returncode,'stdout':prefix+stdout.decode(),'stderr':stderr.decode()}
def observe(p):
    target=p/'tenants/synthetic-process.db'
    with sqlite3.connect(p/'platform.db') as catalog:
        tenant=catalog.execute("SELECT slug,status FROM tenants WHERE slug='synthetic-process'").fetchall()
    observation={'source_sha256':hashfile(p/'source.db'),'original_uploads':{str(f.relative_to(p/'uploads')):hashfile(f) for f in [p/'uploads/a.bin',p/'uploads/nested/b.bin']},
                 'registered':tenant,'target_database_exists':target.exists(),
                 'target_uploads':{str(f.relative_to(p/'uploads/synthetic-process')):hashfile(f) for f in (p/'uploads/synthetic-process').rglob('*') if f.is_file()},
                 'recoveries':[]}
    if target.exists():
        with sqlite3.connect(target) as db:observation['target_integrity']=db.execute('PRAGMA integrity_check').fetchall();observation['target_users']=db.execute('SELECT * FROM users').fetchall();observation['target_records']=db.execute('SELECT * FROM records').fetchall()
    for recovery in sorted((p/'tenants').glob('.adopt-default-*')):
        with sqlite3.connect(recovery/'source.db') as db:integrity=db.execute('PRAGMA integrity_check').fetchall()
        observation['recoveries'].append({'path':str(recovery),'integrity':integrity,'files':sorted(str(f.relative_to(recovery)) for f in recovery.rglob('*') if f.is_file()),'uploads':{str(f.relative_to(recovery/'uploads')):hashfile(f) for f in (recovery/'uploads').rglob('*') if f.is_file()}})
    return observation
receipts=[]
for version,node in versions.items():
    for mode in ['claim','copy']:
        p=fixture();original=hashfile(p/'source.db');original_uploads={str(f.relative_to(p/'uploads')):hashfile(f) for f in [p/'uploads/a.bin',p/'uploads/nested/b.bin']}
        record={'node':version,'mode':mode,'fixture':str(p),'observed_at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'original_source':original,'original_uploads':original_uploads};children=[]
        try:
            if mode=='claim':
                children=[start(node,p,mode),start(node,p,mode)]
                barriers=[await_barrier(c,'before-exclusive-directory-claim') for c in children]
                for c in children:c.stdin.write(b'1');c.stdin.flush()
                results=[collect(c,prefix) for c,prefix in zip(children,barriers)]
                record['processes']=results
                assert sorted(r['exit_code'] for r in results)==[0,1],results
                loser=next(r for r in results if r['exit_code']==1)
                assert 'EEXIST' in loser['stdout'] and 'already exists' not in loser['stdout'],loser
            else:
                child=start(node,p,mode);children=[child]
                prefix=await_barrier(child,'before-second-destination-copy')
                child.send_signal(signal.SIGKILL);killed=collect(child,prefix)
                record['killed_process']=killed
                assert killed['exit_code']==-signal.SIGKILL
                record['before_restart']=observe(p)
                assert record['before_restart']['registered']==[] and not record['before_restart']['target_database_exists']
                assert record['before_restart']['target_uploads']=={'a.bin':original_uploads['a.bin']}
                restarted=start(node,p,'none');children.append(restarted)
                record['restart_process']=collect(restarted)
                assert record['restart_process']['exit_code']==1 and 'Destination already exists' in record['restart_process']['stdout']
            record['after']=observe(p)
            assert record['after']['source_sha256']==original and record['after']['original_uploads']==original_uploads
            assert len(record['after']['recoveries'])==(2 if mode=='claim' else 1)
            for recovery in record['after']['recoveries']:assert recovery['integrity']==[('ok',)] and recovery['uploads']==original_uploads
            if mode=='claim':
                assert record['after']['registered']==[('synthetic-process','active')]
                assert record['after']['target_uploads']==original_uploads and record['after']['target_integrity']==[('ok',)]
                assert record['after']['target_users']==[(1,'synthetic-process-owner','synthetic-original-hash')]
                assert record['after']['target_records']==[(1,'synthetic-original-history')]
            else: assert record['after']['registered']==[] and not record['after']['target_database_exists']
            record['passed']=True
        except Exception as error:
            record['passed']=False;record['failure']=repr(error)
        finally:
            for child in children:
                if child.poll() is None:child.kill();child.wait(timeout=10)
            (out/f'process-{mode}-{version}.json').write_text(json.dumps(record,indent=2)+'\n')
            print(json.dumps({'node':version,'mode':mode,'passed':record['passed'],'failure':record.get('failure')}),flush=True)
            receipts.append(record)
(out/'process-summary.json').write_text(json.dumps([{'node':r['node'],'mode':r['mode'],'passed':r['passed']} for r in receipts],indent=2)+'\n')
assert all(r['passed'] for r in receipts)
