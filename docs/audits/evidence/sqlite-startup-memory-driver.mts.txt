import fs from 'node:fs';

 DATA_DIR:path.join(temp,'tenants'),UPLOAD_DIR:path.join(temp,'uploads'),BACKUP_DIR:path.join(temp,'backups'),PLATFORM_DB_PATH:path.join(temp,'platform.db')});
const d=await import(pathToFileURL(path.join(source,'src/shared/db.ts')).href);const conn=d.getDb();
fs.writeFileSync(result,JSON.stringify({source,healthy:conn.prepare('PRAGMA integrity_check').get(),
 technical_lock_files:fs.readdirSync(temp).filter(n=>n.endsWith('.startup-lock'))},null,2)+'\n');d.closeAllConnections();fs.rmSync(temp,{recursive:true,force:true});
