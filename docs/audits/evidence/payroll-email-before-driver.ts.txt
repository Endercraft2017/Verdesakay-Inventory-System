import {beforeAll,afterAll,afterEach,it,expect} from 'vitest';
import fs from 'node:fs';
import {setupApp,cleanup,login,type TestContext} from '../tests/helpers';
import {smtpCapture} from '../tests/helpers/smtp-capture';
let ctx:TestContext,token:string,peer:Awaited<ReturnType<typeof smtpCapture>>,db:typeof import('../src/shared/db'),form:number,employee:number,user:number,n=0;
const observations:any[]=[];const h=()=>({Authorization:`Bearer ${token}`});
beforeAll(async()=>{ctx=await setupApp();token=await login(ctx.api);db=await import('../src/shared/db');peer=await smtpCapture();
 const {hashPassword}=await import('../src/shared/password-hash');user=Number(db.run("INSERT INTO users(username,full_name,role,password_hash,email) VALUES('mail-proof','Synthetic Mail Recipient','csr',?,'original@example.test')",hashPassword('Synthetic mail proof password')).lastInsertRowid);
 employee=Number(db.run("INSERT INTO employees(employee_no,name,user_id) VALUES('MAIL-PROOF','Synthetic Mail Recipient',?)",user).lastInsertRowid);form=db.queryOne<{id:number}>("SELECT id FROM form_definitions WHERE slug='hr_payroll'")!.id;
 const {saveMailSettings}=await import('../src/shared/mailer');saveMailSettings({transport:'smtp',from_name:'Synthetic Payroll',from_email:'payroll@example.test',reply_to:'',smtp_host:'127.0.0.1',smtp_port:peer.port,smtp_secure:true,smtp_user:''});process.env.NODE_ENV='production';
});
afterEach(()=>{observations.push({messages:JSON.parse(JSON.stringify(peer.messages)),mail_log:db.queryAll('SELECT to_addr,status,provider_id FROM mail_log')});peer.reset();db.run('DELETE FROM mail_log');db.run('DELETE FROM form_records WHERE form_def_id=?',form);db.run("UPDATE users SET email='original@example.test' WHERE id=?",user);});
afterAll(async()=>{process.env.NODE_ENV='test';fs.writeFileSync('work/payroll-email-before-wire.json',JSON.stringify(observations,null,2)+'\n');await peer.close();cleanup(ctx);});
function slip(){const id=Number(db.run("INSERT INTO form_records(form_def_id,number,data,status,created_by) VALUES(?,?,?,'paid',1)",form,'MAIL-PROOF-'+(++n),JSON.stringify({employee_id:employee,employee_name:'Synthetic Mail Recipient',net_pay:1000})).lastInsertRowid);return db.queryOne<any>('SELECT * FROM form_records WHERE id=?',id);}
const send=(id:number,body={})=>ctx.api.post(`/api/hr/payroll/${id}/email`).set(h()).send(body);
it('requires review before provider contact',async()=>{const a=slip(),r=await send(a.id);observations.push({case:'unreviewed',status:r.status,result:r.body});expect(r.status).toBe(409);expect(peer.messages).toHaveLength(0);});
it('rejects stale saved payroll and a recipient changed after staff read',async()=>{const a=slip();db.run("UPDATE form_records SET data=json_set(data,'$.net_pay',2000) WHERE id=?",a.id);db.run("UPDATE users SET email='changed@example.test' WHERE id=?",user);const r=await send(a.id,{expected_revision:a.revision,reviewed_recipient:'original@example.test'});observations.push({case:'stale-review',status:r.status,result:r.body});expect(r.status).toBe(409);expect(peer.messages).toHaveLength(0);});
it('does not send two copies for concurrent submissions of the same payslip',async()=>{const a=slip();const r=await Promise.all([send(a.id),send(a.id)]);observations.push({case:'concurrent',statuses:r.map(r=>r.status)});expect(peer.messages.length).toBeLessThanOrEqual(1);});
it('preserves uncertain SMTP outcome and prevents a blind duplicate retry',async()=>{const a=slip();peer.setMode('disconnect');const first=await send(a.id),second=await send(a.id);observations.push({case:'lost-smtp-ack',statuses:[first.status,second.status],results:[first.body,second.body]});expect(peer.messages.length).toBeLessThanOrEqual(1);expect(db.queryAll("SELECT * FROM mail_log WHERE status='failed'")).toHaveLength(0);});
it('does not lose the accepted outcome when the post-send log write fails',async()=>{const a=slip();db.run("CREATE TRIGGER mail_proof_fail BEFORE INSERT ON mail_log BEGIN SELECT RAISE(ABORT,'synthetic log failure'); END");try{const first=await send(a.id);observations.push({case:'accepted-before-db-failure',status:first.status,messages:peer.messages.length,mail_rows:db.queryAll('SELECT * FROM mail_log').length});expect(first.status).not.toBe(500);expect(peer.messages).toHaveLength(0);}finally{db.run('DROP TRIGGER mail_proof_fail');}});
